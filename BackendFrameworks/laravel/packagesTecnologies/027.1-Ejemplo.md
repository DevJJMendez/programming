- Command

```php
class CreateOrderCommand extends Command
{
    protected $signature = 'make:order';

    protected $description = 'this is a custom command and create a new order';

    public function handle()
    {
        $order = Order::create([
            'user_id' => 10,
            'amount' => 25
        ]);
        $this->info('nuestro comando ha sido ejecutado');
        return Command::SUCCESS;
    }
}
```

- En consola:

```bash
php artisan make:order
```

---

- Pasando parametros:

```php
class CreateOrderCommand extends Command
{

    protected $signature = 'make:order' { user_id} { amount};

    protected $description = 'this is a custom command and create a new order';

    public function handle()
    {
        $user_id = $this->argument('user_id');
        $amount = $this->argument('amount');
        Order::create([
            'user_id' => $user_id,
            'amount' => $amount
        ]);
        $this->info('nuestro comando ha sido ejecutado');
        return Command::SUCCESS;
    }
}
```

- En consola:

```bash
php artisan make:order 66 500
```

## Ejecutar comando desde el controlador

```php
class OrderController extends Controller
{
    public function create()
    {
        Artisan::call(
            'make:order',
            [
                'user_id' => 50,
                'amount' => 900
            ]
        );
        return response()->json("Succesfull");
    }
}
```
